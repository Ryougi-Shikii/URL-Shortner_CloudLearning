
# create sqs queue
aws sqs create-queue --queue-name <producer-consumer-demo>

# list all sqs queues
aws sqs list-queues

==========================================================================================
# producers:
aws sqs send-message \
    --queue-url http://localhost:4566/000000000000/producer-consumer-demo \
    --message-body "<messages>"
#   --message-body "Process video.mp4"


# see attributes of sqs (VisibilityTimeout, ApproximateNumberOfMessages, etc)
aws sqs get-queue-attributes \
    --queue-url http://localhost:4566/000000000000/producer-consumer-demo \
    --attribute-names All

# set long polling time (recommended cuz why not)
aws sqs set-queue-attributes \
    --queue-url http://localhost:4566/000000000000/producer-consumer-demo \
    --attributes ReceiveMessageWaitTimeSeconds=15

==========================================================================================
# receiver
aws sqs receive-message \
    --queue-url http://localhost:4566/000000000000/producer-consumer-demo \
    --max-number-of-messages 3

==========================================================================================
# messages are not deleted (yet)
# They have VisibilityTimeout before messages are made available again to other consumers

# deletion of messages uses (ReceiptHandle by receiver)
aws sqs delete-message \
    --queue-url http://localhost:4566/000000000000/producer-consumer-demo \
    --receipt-handle "<PASTE_YOUR_RECEIPT_HANDLE_HERE>"

===============================================================================================

# setting arn attribute for dlq 
DLQ_ARN=$(aws sqs get-queue-attributes \
    --queue-url http://localhost:4566/000000000000/producer-consumer-dlq \
    --attribute-names QueueArn \
    --query 'Attributes.QueueArn' \
    --output text)
aws sqs set-queue-attributes \
    --queue-url http://localhost:4566/000000000000/producer-consumer-demo \
    --attributes "{\"RedrivePolicy\":\"{\\\"deadLetterTargetArn\\\":\\\"${DLQ_ARN}\\\",\\\"maxReceiveCount\\\":\\\"3\\\"}\"}"
