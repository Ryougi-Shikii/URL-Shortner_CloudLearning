# Create a Table
aws dynamodb create-table \
    --table-name StudentRecords \
    --attribute-definitions AttributeName=StudentID,AttributeType=S \
    --key-schema AttributeName=StudentID,KeyType=HASH \
    --billing-mode PAY_PER_REQUEST

# Insert a Record (Put Item)
aws dynamodb put-item \
    --table-name StudentRecords \
    --item '{
        "StudentID": {"S": "101"},
        "Name": {"S": "Sumit"},
        "Course": {"S": "Cloud Computing"},
        "Score": {"N": "95"}
    }'

# Retrieve the Record (Get Item)
aws dynamodb get-item \
    --table-name StudentRecords \
    --key '{"StudentID": {"S": "101"}}'

# retrive all items
aws dynamodb scan --table-name StudentRecords
